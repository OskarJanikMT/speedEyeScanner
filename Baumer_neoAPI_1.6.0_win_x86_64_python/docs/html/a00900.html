Detailed Description

Cam update callback class to derive from to get informaed about update progress.

import neoapi
// create a class which derives from CamUpdateCallback
class CamUpdateCallbackPrint(neoapi.CamUpdateCallback):
def CurrentUpdateStep(self, step):
print("Current step " + step)
def UpdateProgress(self, progres):
print("Current progress " + str(progres))
def EstimatedTotalUpdateDuration(self, duration):
print("Estimated duration " + str(duration) + " ms")
camera = neoapi.Cam() // create a camera object
camera.Utils.Update.EnableUpdateCallback(callbackPrint); // enable the callback
// -> will be called during update
