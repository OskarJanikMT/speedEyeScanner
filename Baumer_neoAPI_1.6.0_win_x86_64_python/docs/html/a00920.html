16class MyBuffer(neoapi.BufferBase):
17 """implementation for user buffer mode"""
18 def __init__(self, size):
19 neoapi.BufferBase.__init__(self)
20 self._buf = numpy.zeros(size, numpy.uint8) if use_numpy else bytearray(size)
21 self._size = size
22 self.RegisterMemory(self._buf, self._size)
23
24 def CalcVar(self):
25 var = -1
26 if use_numpy: # buf contains the image data and can be used with numpy
27 var = self._buf.var()
28 elif self._size > 0:
29 sum = 0.0
30 for i in range(0, self._size):
31 sum += self._buf[i]
32 avg = sum / self._size
33
34 sum = 0.0
35 for i in range(0, self._size):
36 sum += (self._buf[i] - avg) ** 2
37 var = sum / self._size
38 return var
