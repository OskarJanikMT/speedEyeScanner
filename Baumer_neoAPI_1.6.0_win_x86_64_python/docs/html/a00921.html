18class CamBuffer(neoapi.BufferBase):
19 """implementation for user buffer mode"""
20 def __init__(self, width, height, memtype):
21 neoapi.BufferBase.__init__(self)
22 self.memtype = memtype
23 if memtype == MemoryMode.cpu:
24 self.cpu_mat = numpy.ndarray((height, width, 1), numpy.uint8)

33class EdgeDetector:
34 def __init__(self, serialnumber):
35 self._camera = neoapi.Cam()
36 self._camera.Connect(serialnumber)
37 self._camera.f.ExposureTime.Set(2500)
38 self._camera.DisableChunk()
39 try:
40 self._camera.f.PixelFormat.Set(neoapi.PixelFormat_BayerRG8)
41 except neoapi.FeatureAccessException:
42 self._camera.f.PixelFormat.Set(neoapi.PixelFormat_Mono8)
43 self._pixelformat = self._camera.f.PixelFormat.Get()
44 self._identifier = serialnumber
45 self._buffers = []
46 self._grey_mat = cv2.UMat()
47 self._gauss_mat = cv2.UMat()
48 self._sobel_mat = cv2.UMat()
49 self._detect_thread = threading.Thread()
50 self._frames = 0
51 self._run = False

53 def Setup(self, memtype):
54 cv2.ocl.setUseOpenCL(MemoryMode.cpu != memtype)
55 # if cv2.ocl.Device.getDefault().hostUnifiedMemory():
56 # cv2.ocl.Context.getDefault().setUseSVM(MemoryMode.shared != memtype)
57 self._SetupBuffers(3, memtype)
58 self._camera.SetUserBufferMode()

60 def Detect(self, image, show_image):
61 cambuf = image.GetUserBuffer()
62 img_mat = cambuf.cpu_mat if cambuf.memtype == MemoryMode.cpu else cv2.UMat(cambuf.cpu_mat)
63 if neoapi.PixelFormat_BayerRG8 == self._pixelformat:
64 self._grey_mat = cv2.cvtColor(img_mat, cv2.COLOR_BayerRG2GRAY)
65 else:
66 self._grey_mat = img_mat
67 self._gauss_mat = cv2.GaussianBlur(self._grey_mat, (5, 5), 0)
68 self._sobel_mat = cv2.Sobel(self._gauss_mat, 0, 1, 1, ksize=5)
69 if show_image:
70 cv2.imshow(self._identifier, self._sobel_mat)
71 cv2.pollKey()
72 self._frames += 1
73
74 def ProcessedFrames(self):
75 frames = self._frames
76 self._frames = 0
77 return frames
78
79 def GetIdentifier(self):
80 return self._identifier

82 def Start(self, show_images):
83 self._run = True
84 self._detect_thread = threading.Thread(target=self._Detect, args=(show_images,))
85 self._detect_thread.start()

87 def Stop(self):
88 self._run = False
89 if self._detect_thread.is_alive():

92 def FreeCamBuffers(self):
93 while self._buffers:
94 self._camera.RevokeUserBuffer(self._buffers.pop())

96 def _SetupBuffers(self, count, memtype):
97 width = self._camera.f.Width.Get()
98 height = self._camera.f.Height.Get()
99 self.FreeCamBuffers()
100 for _ in range(count):
101 self._buffers.append(CamBuffer(width, height, memtype))
102 self._camera.AddUserBuffer(self._buffers[-1])
103 self._grey_mat = cv2.UMat(height, width, cv2.CV_8UC1, memtype)
104 self._gauss_mat = cv2.UMat(height, width, cv2.CV_8UC1, memtype)
105 self._sobel_mat = cv2.UMat(height, width, cv2.CV_8UC1, memtype)
106
107 def _Detect(self, show_images):
108 try:
109 if show_images:
110 cv2.namedWindow(self._identifier, cv2.WINDOW_NORMAL)
111 while self._run:
112 image = self._camera.GetImage()
113 if image.IsEmpty():
114 print("%s Error during acquisition!" % self._identifier)
115 break
116 self.Detect(image, show_images)
117 if show_images:
118 cv2.destroyWindow(self._identifier)
119 except neoapi.NeoException as exc:
120 print("%s error %s" % (self._identifier, exc.GetDescription()))
121 except cv2.error as exc:
122 print("%s cv error %s" % (self._identifier, exc))

133def FindDevices():
134 devices = []
135 for device in neoapi.CamInfoList.Get():
136 try:
137 devices.append(EdgeDetector(device.GetSerialNumber()))

149def RunDetection(devices, memtypes, show_images):
150 if devices:
151 for val, text in memtypes.items():
152 print("Next run will be processed on", text)
153 for device in devices:
154 device.Setup(val)
155 for device in devices:
156 device.Start(show_images)
157 PrintMetrics(devices, 5)
158 for device in devices:
159 device.Stop()

161def FreeDevices(devices):
